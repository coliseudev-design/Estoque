SELECT PEDIDOS.ID_PEDIDO, PEDIDOS.PEDIDO, PEDIDOS.DAV, PEDIDOS.NPV, PEDIDOS.ID_CAIXA, PEDIDOS.DATA_HORA, PEDIDOS.DATA_VENCIMENTO, PEDIDOS.DATA_ENTREGA, PEDIDOS.NF, PEDIDO_ITENS.ID_NOTA, PEDIDOS.NOTA_FISCAL, PEDIDOS.NOTA_FISCAL_SERV, PEDIDOS.CUPOM_FISCAL, PEDIDOS.NFCE_NUMERO,
PEDIDOS.ID_CLIENTE, PEDIDOS.DESCONTO, PEDIDO_ITENS.DESCONTO, PEDIDOS.ACRESCIMO, PEDIDOS.ID_FUNCIONARIO, PEDIDOS.ID_VENDEDOR, CLIENTES.ID_VENDEDOR, PEDIDOS.ID_NATUREZA, PEDIDOS.ID_FORMA, PEDIDOS.ID_ESPECIE, PEDIDOS.ID_PROPRIEDADE, PEDIDOS.STATUS, PEDIDOS.ID_REFERENCIA, CLIENTES.ID_CLASSE,
PEDIDOS.ID_TRANSPORTADORA, PEDIDOS.TIPO, NATUREZA_OPERACAO.OPERACAO, PEDIDOS.CONDICIONAL, PEDIDOS.id_depto, PEDIDOS.iss, PEDIDO_ITENS.TIPO, PEDIDOS.ID_VEICULO, PEDIDOS.PLACA_VEICULO, PEDIDOS.ID_EQUIPAMENTO, PEDIDO_ITENS.ID_TECNICO, PEDIDO_ITENS.ID_PRODUTO, PEDIDO_ITENS.CODIGO_BARRA, PEDIDO_ITENS.DESCRICAO,
(CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.qtde WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.qtde*-1) END),
PEDIDO_ITENS.VALOR_FINAL_UN, PEDIDO_ITENS.VALOR_UNITARIO, PEDIDO_ITENS.VALOR_BASE, PEDIDO_ITENS.VALOR_CUSTO, PEDIDO_ITENS.VALOR_MINIMO,
CAST ((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END) -
     (((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END)*PEDIDOS.DESCONTO)/100) -
     (CASE WHEN (CLIENTES.substituto = 1) AND ((PEDIDO_ITENS.tipo = 3) OR (PEDIDO_ITENS.tipo = 6)) THEN (((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END)*PEDIDOS.ISS)/100) ELSE 0 END) -
     (CASE WHEN COALESCE(PEDIDOS.IRRF,0) > 0 THEN ((PEDIDOS.VALOR_SERVICOS*PEDIDOS.IRRF)/100) ELSE 0 END) AS DECIMAL(15,4)),
CAST ((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END) -
     (((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END)*PEDIDO_ITENS.DESCONTO)/100) -
     (CASE WHEN (CLIENTES.substituto = 1) AND ((PEDIDO_ITENS.tipo = 3) OR (PEDIDO_ITENS.tipo = 6)) THEN (((CASE WHEN (NATUREZA_OPERACAO.es = 1) THEN PEDIDO_ITENS.valor_final WHEN (NATUREZA_OPERACAO.es = 2) THEN (PEDIDO_ITENS.valor_final*-1) END)*PEDIDOS.ISS)/100) ELSE 0 END) -
     (CASE WHEN COALESCE(PEDIDOS.IRRF,0) > 0 THEN ((PEDIDOS.VALOR_SERVICOS*PEDIDOS.IRRF)/100) ELSE 0 END) AS DECIMAL(15,4)),
PEDIDO_ITENS.ICMS, PEDIDO_ITENS.IPI, PEDIDO_ITENS.DADOS, PEDIDO_ITENS.CFOP, PEDIDO_ITENS.base_pis, PEDIDO_ITENS.PIS, PEDIDO_ITENS.base_cofins, PEDIDO_ITENS.COFINS
FROM PEDIDOS
INNER JOIN PEDIDO_ITENS ON (PEDIDO_ITENS.ID_PEDIDO = PEDIDOS.ID_PEDIDO)
LEFT JOIN NATUREZA_OPERACAO ON (NATUREZA_OPERACAO.ID_NATUREZA = PEDIDOS.ID_NATUREZA)
LEFT JOIN CLIENTES ON (CLIENTES.ID_CLIENTE = PEDIDOS.ID_CLIENTE)
WHERE ((PEDIDO_ITENS.status = 1) OR (PEDIDO_ITENS.status IS NULL))
