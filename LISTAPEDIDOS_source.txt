SELECT PEDIDOS.ID_PEDIDO, PEDIDOS.PEDIDO, PEDIDOS.DAV, PEDIDOS.NPV, PEDIDOS.ID_CAIXA, PEDIDOS.DATA_HORA, PEDIDOS.DATA_VENCIMENTO, PEDIDOS.DATA_ENTREGA, PEDIDOS.NOTA_FISCAL, PEDIDOS.NOTA_FISCAL_SERV, PEDIDOS.CUPOM_FISCAL, PEDIDOS.NFCE_NUMERO, PEDIDOS.ID_CLIENTE,
PEDIDOS.DESCONTO, PEDIDOS.ACRESCIMO, PEDIDOS.ID_FUNCIONARIO, PEDIDOS.ID_VENDEDOR, CLIENTES.ID_VENDEDOR, PEDIDOS.ID_NATUREZA, PEDIDOS.ID_FORMA, PEDIDOS.ID_ESPECIE, PEDIDOS.ID_PROPRIEDADE, PEDIDOS.STATUS, PEDIDOS.TIPO, CLIENTES.ID_CLASSE,
NATUREZA_OPERACAO.OPERACAO, NATUREZA_OPERACAO.PROCESSO, NATUREZA_OPERACAO.TIPO, PEDIDOS.CONDICIONAL, PEDIDOS.ID_DEPTO, PEDIDOS.ID_VEICULO, PEDIDOS.PLACA_VEICULO, PEDIDOS.ID_REFERENCIA, PEDIDOS.ID_TRANSPORTADORA, PEDIDOS.VALOR_DESCONTO, PEDIDOS.VALOR_ACRESCIMO,
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN PEDIDOS.VALOR_PEDIDO ELSE (PEDIDOS.VALOR_PEDIDO*-1) END),
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN PEDIDOS.VALOR_SERVICOS ELSE (PEDIDOS.VALOR_SERVICOS*-1) END),
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN PEDIDOS.VALOR_BASE ELSE (PEDIDOS.VALOR_BASE*-1) END),
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN PEDIDOS.VALOR_CUSTOS ELSE (PEDIDOS.VALOR_CUSTOS*-1) END),
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN PEDIDOS.VALOR_FRETE ELSE (PEDIDOS.VALOR_FRETE*-1) END),
(CASE WHEN ((natureza_operacao.es = 1) AND (pedidos.tipo = 1)) OR ((natureza_operacao.es = 2) AND (pedidos.tipo = 2)) THEN
      ((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS) - CASE WHEN PEDIDOS.VALOR_DESCONTO IS NOT NULL THEN PEDIDOS.VALOR_DESCONTO ELSE (((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS)*PEDIDOS.DESCONTO)/100) END + CASE WHEN PEDIDOS.VALOR_ACRESCIMO IS NOT NULL THEN PEDIDOS.VALOR_ACRESCIMO ELSE (((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS)*PEDIDOS.ACRESCIMO)/100) END + (CASE WHEN (PEDIDOS.TIPO_FRETE = 2) THEN PEDIDOS.VALOR_FRETE ELSE 0 END) + CASE WHEN PEDIDOS.OPE_SOMA_OD = 1 THEN PEDIDOS.OUTRAS_DESPESAS ELSE 0 END + PEDIDOS.VALOR_IPI - (CASE WHEN clientes.substituto = 1 THEN (( (PEDIDOS.VALOR_SERVICOS - ((PEDIDOS.VALOR_SERVICOS*COALESCE(PEDIDOS.DESCONTOSV,0))/100)) *(CASE WHEN PEDIDOS.ISS IS NULL THEN 5 ELSE PEDIDOS.ISS END))/100) ELSE 0 END) - (CASE WHEN COALESCE(PEDIDOS.IRRF,0) > 0 THEN ((PEDIDOS.VALOR_SERVICOS*PEDIDOS.IRRF)/100) ELSE 0 END) )
      ELSE
      (((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS) - CASE WHEN PEDIDOS.VALOR_DESCONTO IS NOT NULL THEN PEDIDOS.VALOR_DESCONTO ELSE (((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS)*PEDIDOS.DESCONTO)/100) END + CASE WHEN PEDIDOS.VALOR_ACRESCIMO IS NOT NULL THEN PEDIDOS.VALOR_ACRESCIMO ELSE (((PEDIDOS.VALOR_PEDIDO+PEDIDOS.VALOR_SERVICOS)*PEDIDOS.ACRESCIMO)/100) END + (CASE WHEN (PEDIDOS.TIPO_FRETE = 2) THEN PEDIDOS.VALOR_FRETE ELSE 0 END) + CASE WHEN PEDIDOS.OPE_SOMA_OD = 1 THEN PEDIDOS.OUTRAS_DESPESAS ELSE 0 END + PEDIDOS.VALOR_IPI - (CASE WHEN clientes.substituto = 1 THEN (( (PEDIDOS.VALOR_SERVICOS - ((PEDIDOS.VALOR_SERVICOS*COALESCE(PEDIDOS.DESCONTOSV,0))/100)) *(CASE WHEN PEDIDOS.ISS IS NULL THEN 5 ELSE PEDIDOS.ISS END))/100) ELSE 0 END) - (CASE WHEN COALESCE(PEDIDOS.IRRF,0) > 0 THEN ((PEDIDOS.VALOR_SERVICOS*PEDIDOS.IRRF)/100) ELSE 0 END) )*-1) END)
FROM PEDIDOS
LEFT JOIN CLIENTES ON (CLIENTES.ID_CLIENTE = PEDIDOS.ID_CLIENTE)
LEFT JOIN NATUREZA_OPERACAO ON (NATUREZA_OPERACAO.ID_NATUREZA = PEDIDOS.ID_NATUREZA)
